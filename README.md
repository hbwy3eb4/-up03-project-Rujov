## 🔑 ШПАРГАЛКА ПО SQL
-- Выборка
SELECT * FROM Товар;                        -- все поля
SELECT название, цена FROM Товар;           -- конкретные
SELECT DISTINCT категория FROM Товар;       -- уникальные

-- Фильтрация
WHERE цена > 5000                           -- больше
WHERE количество <= 3                       -- меньше или равно
WHERE категория = 'Обувь'                   -- равно
WHERE название LIKE '%о%'                   -- поиск подстроки
WHERE цена BETWEEN 1000 AND 5000            -- диапазон

-- Сортировка
ORDER BY цена ASC                           -- возрастание
ORDER BY цена DESC                          -- убывание
LIMIT 1                                     -- только 1 строка

-- Агрегация
COUNT(*)                                    -- количество
SUM(цена * количество)                      -- сумма
AVG(цена)                                   -- среднее
MIN(цена)                                   -- минимум
MAX(цена)                                   -- максимум

-- Группировка
GROUP BY клиент                             -- группировка
GROUP BY клиент HAVING COUNT(*) > 1         -- с условием

-- Соединение
FROM Заказ
JOIN Товар ON Заказ.товар_id = Товар.id     -- JOIN
